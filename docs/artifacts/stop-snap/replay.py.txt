#!/usr/bin/env python3
"""Offline replay for the Foot IK T3 release-catch-up study.

This intentionally uses only the Python standard library.  It replays horizontal
T3 state, verifies R0 against the recorder, and then evaluates parameterized
candidate policies against R0 lock episodes.
"""

import argparse
import csv
import math
import statistics
import sys
from dataclasses import dataclass, field
from pathlib import Path


DT = 1.0 / 60.0
SETTINGS = {
    "FootGroundedHeightMin": 0.08,
    "FootGroundedHeightMax": 0.25,
    "ContactPhaseSpeedMin": 1.0,
    "ContactPhaseSpeedMax": 4.0,
    "LockBeta": 0.95,
    "UnlockBeta": 0.7,
    "UnlockRadius": 0.2,
    "ReleaseTime": 0.12,
    "MaxPoseStepPerFrame": 0.5,
}

REQUIRED_COLUMNS = {
    "frame", "phase", "gait", "stopBake",
    "rootX", "rootZ", "rootEvalX", "rootEvalZ", "posePlaneY",
}
for _side in ("l", "r"):
    REQUIRED_COLUMNS.update({
        _side + "ph", _side + "poseX", _side + "poseY", _side + "poseZ",
        _side + "tgtX", _side + "tgtZ", _side + "stance", _side + "prevValid",
    })


def finite(value):
    # 未匹配到候選事件的欄位是空字串（""）⇒ 視為缺值
    return isinstance(value, (int, float)) and math.isfinite(value)


def number(row, name):
    try:
        return float(row[name])
    except (KeyError, TypeError, ValueError) as exc:
        raise ValueError("invalid numeric value for column %s: %r" % (name, row.get(name))) from exc


def flag(row, name):
    value = row[name].strip().lower()
    if value in ("1", "true"):
        return True
    if value in ("0", "false"):
        return False
    raise ValueError("invalid boolean value for column %s: %r" % (name, row[name]))


def distance2(a, b):
    return math.hypot(a[0] - b[0], a[1] - b[1])


def distance3(a, b):
    return math.sqrt(sum((a[i] - b[i]) ** 2 for i in range(3)))


def smoothstep01(value):
    value = min(1.0, max(0.0, value))
    return value * value * (3.0 - 2.0 * value)


def percentile(values, q):
    values = sorted(v for v in values if finite(v))
    if not values:
        return math.nan
    if len(values) == 1:
        return values[0]
    position = (len(values) - 1) * q
    lower = int(math.floor(position))
    upper = int(math.ceil(position))
    if lower == upper:
        return values[lower]
    fraction = position - lower
    return values[lower] * (1.0 - fraction) + values[upper] * fraction


def median(values):
    values = [v for v in values if finite(v)]
    return statistics.median(values) if values else math.nan


def maximum(values):
    values = [v for v in values if finite(v)]
    return max(values) if values else math.nan


def fmt(value, digits=3):
    if not finite(value):
        return ""
    return ("%%.%df" % digits) % value


def md(value, digits=2):
    return fmt(value, digits) if finite(value) else "—"


@dataclass
class CaseData:
    path: Path
    name: str
    rows: list
    gait: str
    terrain: str
    stop_start: int

    @property
    def stop_end(self):
        return len(self.rows) - 1

    def stop_bake(self, index):
        if 0 <= index < len(self.rows) and self.rows[index]["stopBake"]:
            return self.rows[index]["stopBake"]
        for i in range(min(index, self.stop_end), self.stop_start - 1, -1):
            if self.rows[i]["stopBake"]:
                return self.rows[i]["stopBake"]
        for i in range(self.stop_start, self.stop_end + 1):
            if self.rows[i]["stopBake"]:
                return self.rows[i]["stopBake"]
        return ""


@dataclass(frozen=True)
class Policy:
    name: str
    unlock_radius: float = SETTINGS["UnlockRadius"]
    trend_n: int = 0
    trend_d0: float = 0.0
    follow_d0: float = 0.0
    slow_k: int = 0
    slow_v: float = 0.0
    settle: bool = False
    cap_k: float = 0.0
    cap_v_min: float = 0.5


@dataclass
class ReplayState:
    phase: int = 0
    has_history: bool = False
    prev_pose: tuple = (0.0, 0.0, 0.0)
    prev_height: float = 0.0
    prev_valid: bool = False
    lock: tuple = (0.0, 0.0)
    release_blend: float = 0.0
    release_offset: tuple = (0.0, 0.0)
    trend_last_gap: float = 0.0
    trend_count: int = 0
    speeds: list = field(default_factory=list)
    prev_predicted: tuple = None
    prev_horizontal: tuple = None


@dataclass
class FootReplay:
    phase: list
    beta: list
    speed: list
    predicted: list
    horizontal: list
    animation: list
    screen: list
    episodes: list


@dataclass
class Episode:
    lock: int
    release: object
    locked_indices: list

    @property
    def lock_length(self):
        return len(self.locked_indices)


def policy_registry():
    policies = {"R0": Policy("R0")}
    for radius in (0.10, 0.06, 0.04, 0.03):
        name = "R1a-r%.2f" % radius
        policies[name] = Policy(name, unlock_radius=radius)
    for count in (3, 5):
        for distance in (0.02, 0.03):
            name = "R1b-trend-N%d-d%.2f" % (count, distance)
            policies[name] = Policy(name, trend_n=count, trend_d0=distance)
    for distance in (0.02, 0.03, 0.05):
        name = "R1c-follow-d%.2f" % distance
        policies[name] = Policy(name, follow_d0=distance)
    for count in (2, 3):
        for speed in (0.3, 0.5):
            name = "R2a-slow-K%d-v%.1f" % (count, speed)
            policies[name] = Policy(name, slow_k=count, slow_v=speed)
    policies["R2b-settle-K3"] = Policy("R2b-settle-K3", settle=True)
    for multiplier in (1.0, 1.25):
        name = "R3-cap-k%.2f-v0.5" % multiplier
        policies[name] = Policy(name, cap_k=multiplier)
    return policies


def combine_policy(spec, registry):
    parts = spec.split(":")
    if len(parts) != 2 or not parts[0].startswith("R2") or not parts[1].startswith("R1"):
        raise ValueError("combo must be R2x:R1y (for example R2a-slow-K3-v0.5:R1a-r0.06)")
    if parts[0] not in registry or parts[1] not in registry:
        raise ValueError("unknown combo member: %s" % spec)
    r2, r1 = registry[parts[0]], registry[parts[1]]
    return Policy(
        spec,
        unlock_radius=r1.unlock_radius,
        trend_n=r1.trend_n,
        trend_d0=r1.trend_d0,
        follow_d0=r1.follow_d0,
        slow_k=r2.slow_k,
        slow_v=r2.slow_v,
        settle=r2.settle,
    )


def load_cases(input_dir):
    paths = sorted(input_dir.glob("snap_on_*.csv"))
    if not paths:
        raise ValueError("no snap_on_*.csv files found in %s" % input_dir)
    cases = []
    errors = []
    for path in paths:
        with path.open("r", encoding="utf-8-sig", newline="") as stream:
            reader = csv.DictReader(stream)
            fields = set(reader.fieldnames or [])
            missing = sorted(REQUIRED_COLUMNS - fields)
            if missing:
                errors.append("%s: missing required replay columns: %s" % (path.name, ", ".join(missing)))
                continue
            rows = list(reader)
        restart = next((i for i, row in enumerate(rows) if row["phase"] == "restart"), len(rows))
        rows = rows[:restart]
        if not rows:
            errors.append("%s: no rows before restart" % path.name)
            continue
        stop_start = next((i for i, row in enumerate(rows) if row["phase"] == "stop"), None)
        if stop_start is None:
            errors.append("%s: no stop phase" % path.name)
            continue
        frames = [int(float(row["frame"])) for row in rows]
        gaps = [(frames[i - 1], frames[i]) for i in range(1, len(frames)) if frames[i] != frames[i - 1] + 1]
        if gaps:
            errors.append("%s: recording is not contiguous (first gap %d -> %d)" % (path.name, gaps[0][0], gaps[0][1]))
            continue
        terrain = "flat" if "_flat_" in path.stem else "Up" if "_Up_" in path.stem else "Down" if "_Down_" in path.stem else "unknown"
        gait = rows[0]["gait"]
        if terrain == "unknown":
            errors.append("%s: filename does not encode terrain as flat/Up/Down" % path.name)
            continue
        if gait not in ("Walk", "Run"):
            errors.append("%s: unsupported gait %r" % (path.name, gait))
            continue
        if not any(row["stopBake"] for row in rows[stop_start:]):
            errors.append("%s: stop phase has no stopBake name" % path.name)
            continue
        cases.append(CaseData(path, path.stem, rows, gait, terrain, stop_start))
    if errors:
        raise ValueError("input validation failed; replay requires newly recorded stance/prevValid data:\n  " + "\n  ".join(errors))
    return cases


def lock_gate(policy, speeds):
    if policy.slow_k:
        return len(speeds) >= policy.slow_k and all(v < policy.slow_v for v in speeds[-policy.slow_k:])
    if policy.settle:
        recent = speeds[-3:]
        return len(recent) == 3 and all(v < 1.0 for v in recent) and all(recent[i] <= recent[i - 1] + 0.05 for i in range(1, 3))
    return True


def release_position(predicted, offset, blend):
    weight = smoothstep01(blend)
    return predicted[0] + weight * offset[0], predicted[1] + weight * offset[1]


def cap_release(state, predicted, desired, blend_before, policy):
    if state.prev_horizontal is None or state.prev_predicted is None:
        return desired
    animation_step = distance2(predicted, state.prev_predicted)
    limit = max(policy.cap_k * animation_step, policy.cap_v_min * DT)
    if distance2(desired, state.prev_horizontal) <= limit + 1e-12:
        return desired

    low = state.release_blend if state.phase == 2 else 0.0
    high = max(low, blend_before)
    high_position = release_position(predicted, state.release_offset, high)
    if distance2(high_position, state.prev_horizontal) > limit + 1e-9:
        # With k >= 1 this should only be reachable through a discontinuity.  Keep
        # the most retained valid blend; R0 correctness is independent of R3.
        state.phase = 2
        state.release_blend = high
        return high_position
    for _ in range(48):
        middle = (low + high) * 0.5
        position = release_position(predicted, state.release_offset, middle)
        if distance2(position, state.prev_horizontal) <= limit:
            high = middle
        else:
            low = middle
    state.release_blend = high
    state.phase = 2 if high > 1e-8 else 0
    return release_position(predicted, state.release_offset, high)


def replay_foot(case, side, policy):
    state = ReplayState()
    phases, betas, speeds, predicted_values, horizontal_values = [], [], [], [], []
    for row in case.rows:
        pose = (number(row, side + "poseX"), number(row, side + "poseY"), number(row, side + "poseZ"))
        pose_height = pose[1] - number(row, "posePlaneY")
        continuous = state.has_history and distance3(pose, state.prev_pose) <= SETTINGS["MaxPoseStepPerFrame"]
        if continuous:
            step = tuple(pose[i] - state.prev_pose[i] for i in range(3))
            predicted = (pose[0] + step[0], pose[2] + step[2])
            speed = math.hypot(step[0], step[2]) / DT
        else:
            predicted = (pose[0], pose[2])
            speed = 0.0
        state.has_history = True
        state.prev_pose = pose
        state.prev_height = pose_height
        if not continuous:
            state.phase = 0
            state.release_blend = 0.0
            state.prev_valid = False
            state.trend_count = 0
            state.trend_last_gap = 0.0

        state.speeds.append(speed)
        if len(state.speeds) > 5:
            del state.speeds[0]
        t = ((speed - SETTINGS["ContactPhaseSpeedMin"]) /
             (SETTINGS["ContactPhaseSpeedMax"] - SETTINGS["ContactPhaseSpeedMin"]))
        speed_phase = 1.0 - smoothstep01(t)
        beta = min(1.0, max(0.0, number(row, side + "stance"))) * speed_phase

        phase_before = state.phase
        blend_before = state.release_blend
        if state.phase == 0:
            if beta >= SETTINGS["LockBeta"] and state.prev_valid and lock_gate(policy, state.speeds):
                state.phase = 1
                state.lock = predicted
                state.trend_count = 0
                state.trend_last_gap = 0.0
        elif state.phase == 1:
            gap_vector = (predicted[0] - state.lock[0], predicted[1] - state.lock[1])
            gap = math.hypot(gap_vector[0], gap_vector[1])
            if policy.follow_d0 > 0.0 and gap > policy.follow_d0:
                scale = 1.0 - policy.follow_d0 / gap
                state.lock = (state.lock[0] + gap_vector[0] * scale, state.lock[1] + gap_vector[1] * scale)
                gap_vector = (predicted[0] - state.lock[0], predicted[1] - state.lock[1])
                gap = math.hypot(gap_vector[0], gap_vector[1])
            trend_release = False
            if policy.trend_n:
                if gap > state.trend_last_gap + 1e-9:
                    state.trend_count += 1
                else:
                    state.trend_count = 0
                trend_release = state.trend_count >= policy.trend_n and gap > policy.trend_d0
                state.trend_last_gap = gap
            if (beta < SETTINGS["UnlockBeta"] or gap > policy.unlock_radius or
                    not state.prev_valid or trend_release):
                state.phase = 2
                state.release_blend = 1.0
                state.release_offset = (-gap_vector[0], -gap_vector[1])
        else:
            state.release_blend = (state.release_blend - DT / SETTINGS["ReleaseTime"]
                                   if SETTINGS["ReleaseTime"] > 0.0 else 0.0)
            if state.release_blend <= 0.0:
                state.release_blend = 0.0
                state.phase = 0

        if state.phase == 1:
            horizontal = state.lock
        elif state.phase == 2:
            horizontal = release_position(predicted, state.release_offset, state.release_blend)
        else:
            horizontal = predicted
        if policy.cap_k > 0.0 and (phase_before == 2 or state.phase == 2):
            horizontal = cap_release(state, predicted, horizontal, blend_before, policy)

        phases.append(state.phase)
        betas.append(beta)
        speeds.append(speed)
        predicted_values.append(predicted)
        horizontal_values.append(horizontal)
        state.prev_valid = flag(row, side + "prevValid")
        state.prev_predicted = predicted
        state.prev_horizontal = horizontal

    animation = []
    screen = [None]
    for i, row in enumerate(case.rows):
        delta = (number(row, "rootX") - number(row, "rootEvalX"),
                 number(row, "rootZ") - number(row, "rootEvalZ"))
        animation.append((number(row, side + "poseX") + delta[0], number(row, side + "poseZ") + delta[1]))
        if i:
            screen.append((horizontal_values[i - 1][0] + delta[0], horizontal_values[i - 1][1] + delta[1]))
    return FootReplay(phases, betas, speeds, predicted_values, horizontal_values, animation, screen,
                      find_episodes(phases))


def find_episodes(phases):
    episodes = []
    i = 0
    while i < len(phases):
        if phases[i] == 1 and (i == 0 or phases[i - 1] != 1):
            lock = i
            j = i
            while j < len(phases) and phases[j] == 1:
                j += 1
            release = j if j < len(phases) and phases[j] == 2 else None
            episodes.append(Episode(lock, release, list(range(lock, j))))
            i = j
        else:
            i += 1
    return episodes


def path_length(track, start, end):
    if start is None or end is None or end <= start:
        return 0.0
    total = 0.0
    used = False
    for i in range(start + 1, min(end, len(track) - 1) + 1):
        if track[i] is None or track[i - 1] is None:
            continue
        total += distance2(track[i], track[i - 1]) * 100.0
        used = True
    return total if used else math.nan


def max_step(track, start, end):
    values = []
    for i in range(start + 1, min(end, len(track) - 1) + 1):
        if track[i] is not None and track[i - 1] is not None:
            values.append(distance2(track[i], track[i - 1]) * 100.0)
    return max(values) if values else math.nan


def jerk_values(track):
    values = []
    for i in range(2, len(track)):
        if track[i] is None or track[i - 1] is None or track[i - 2] is None:
            continue
        dx = track[i][0] - 2.0 * track[i - 1][0] + track[i - 2][0]
        dz = track[i][1] - 2.0 * track[i - 1][1] + track[i - 2][1]
        values.append(math.hypot(dx, dz) * 100.0)
    return values


def episode_windows(episodes, length):
    windows = []
    for i, episode in enumerate(episodes):
        lower = 0 if i == 0 else (episodes[i - 1].lock + episode.lock) // 2 + 1
        upper = length - 1 if i + 1 == len(episodes) else (episode.lock + episodes[i + 1].lock) // 2
        windows.append((lower, upper))
    return windows


def match_episode(candidates, lower, upper, target_lock):
    choices = [episode for episode in candidates if lower <= episode.lock <= upper]
    return min(choices, key=lambda episode: abs(episode.lock - target_lock)) if choices else None


def landing_metrics(track, animation, lock):
    if lock < 6 or lock + 4 >= len(track) or any(track[i] is None for i in range(lock - 3, lock + 5)):
        return math.nan, math.nan
    rest = animation[lock + 3]
    direction = (rest[0] - animation[lock - 6][0], rest[1] - animation[lock - 6][1])
    length = math.hypot(direction[0], direction[1])
    if length <= 1e-9:
        return math.nan, math.nan
    direction = (direction[0] / length, direction[1] / length)
    project = lambda point: ((point[0] - rest[0]) * direction[0] + (point[1] - rest[1]) * direction[1]) * 100.0
    overshoot = max(project(track[i]) for i in range(lock - 3, lock + 4))
    pullback = overshoot - project(track[lock + 4])
    return overshoot, pullback


def event_metadata(case, side, episodes):
    stop_ordinal = 0
    release_indices = [episode.release for episode in episodes if episode.release is not None and episode.release >= case.stop_start]
    last_release = max(release_indices) if release_indices else None
    metadata = []
    for index, episode in enumerate(episodes, 1):
        if episode.lock >= case.stop_start:
            stop_ordinal += 1
            ordinal = stop_ordinal
        else:
            ordinal = 0
        anchor = episode.release if episode.release is not None else episode.lock
        context = "loop" if anchor < case.stop_start else case.stop_bake(anchor)
        metadata.append({
            "eventId": "%s:%s:%03d" % (case.name, side, index),
            "case": case.name,
            "gait": case.gait,
            "terrain": case.terrain,
            "side": side.upper(),
            "context": context or "",
            "ordinal": ordinal,
            "tRel": episode.release - case.stop_start if episode.release is not None else math.nan,
            "isLastRelease": int(episode.release is not None and episode.release == last_release),
            "r0LockFrame": int(float(case.rows[episode.lock]["frame"])),
            "r0ReleaseFrame": int(float(case.rows[episode.release]["frame"])) if episode.release is not None else "",
        })
    return metadata


EVENT_FIELDS = [
    "policy", "eventId", "case", "gait", "terrain", "side", "context", "ordinal", "tRel", "isLastRelease",
    "r0LockFrame", "r0ReleaseFrame", "policyLockFrame", "policyReleaseFrame",
    "m1GapCm", "m2PathRatio", "m2PeakRatio", "m2ScreenPathCm", "m2AnimPathCm",
    "m3EndGapCm", "m3LatePathRatio", "m3ScreenPathCm", "m3AnimPathCm",
    "m4LockedPathCm", "m4LockedMaxStepCm",
    "m5OvershootCm", "m5PullbackCm", "m5LockFrameDelta",
    "m6LockCount", "m6LockedFrames", "m6LockLenMedian", "m6LockLenMax",
    "m7P95CmPerFrame2", "m7MaxCmPerFrame2",
]


def evaluate_policy(cases, r0_results, policy):
    event_rows = []
    aggregate = {
        "m1_loop": [], "m1_stop": [], "m2_loop": [], "m2_stop": [],
        "m2peak_loop": [], "m2peak_stop": [], "m4_loop": [], "m4_stop": [],
        "m4peak_loop": [], "m4peak_stop": [], "m5": [], "m3gap": [], "m3ratio": [],
        "jerk": [], "lock_count": 0, "locked_frames": 0, "lock_lengths": [],
    }
    for case in cases:
        current = {side: replay_foot(case, side, policy) for side in ("l", "r")}
        for side in ("l", "r"):
            candidate = current[side]
            baseline = r0_results[case.name][side]
            aggregate["lock_count"] += len(candidate.episodes)
            aggregate["locked_frames"] += sum(episode.lock_length for episode in candidate.episodes)
            aggregate["lock_lengths"].extend(episode.lock_length for episode in candidate.episodes)
            case_jerk = jerk_values(candidate.screen)
            aggregate["jerk"].extend(case_jerk)
            case_p95 = percentile(case_jerk, 0.95)
            case_max = maximum(case_jerk)
            windows = episode_windows(baseline.episodes, len(case.rows))
            metadata = event_metadata(case, side, baseline.episodes)
            for baseline_episode, (lower, upper), meta in zip(baseline.episodes, windows, metadata):
                matched = match_episode(candidate.episodes, lower, upper, baseline_episode.lock)
                in_window = [episode for episode in candidate.episodes if lower <= episode.lock <= upper]
                row = {field: "" for field in EVENT_FIELDS}
                row.update(meta)
                row["policy"] = policy.name
                row["m7P95CmPerFrame2"] = case_p95
                row["m7MaxCmPerFrame2"] = case_max
                row["m6LockCount"] = len(in_window)
                lengths = [episode.lock_length for episode in in_window]
                row["m6LockedFrames"] = sum(lengths)
                row["m6LockLenMedian"] = median(lengths)
                row["m6LockLenMax"] = maximum(lengths)

                locked = baseline_episode.locked_indices
                if locked:
                    row["m4LockedPathCm"] = path_length(candidate.screen, locked[0], locked[-1])
                    row["m4LockedMaxStepCm"] = max_step(candidate.screen, locked[0], locked[-1])
                    group = "loop" if meta["context"] == "loop" else "stop"
                    aggregate["m4_" + group].append(row["m4LockedPathCm"])
                    aggregate["m4peak_" + group].append(row["m4LockedMaxStepCm"])

                if matched is not None:
                    row["policyLockFrame"] = int(float(case.rows[matched.lock]["frame"]))
                    row["m5LockFrameDelta"] = matched.lock - baseline_episode.lock
                    overshoot, pullback = landing_metrics(candidate.screen, candidate.animation, matched.lock)
                    row["m5OvershootCm"] = overshoot
                    row["m5PullbackCm"] = pullback
                    aggregate["m5"].append(overshoot)
                    if matched.release is not None:
                        release = matched.release
                        row["policyReleaseFrame"] = int(float(case.rows[release]["frame"]))
                        if candidate.screen[release] is not None:
                            row["m1GapCm"] = distance2(candidate.screen[release], candidate.animation[release]) * 100.0
                        if release + 10 < len(case.rows):
                            screen_path = path_length(candidate.screen, release, release + 10)
                            animation_path = path_length(candidate.animation, release, release + 10)
                            row["m2ScreenPathCm"] = screen_path
                            row["m2AnimPathCm"] = animation_path
                            row["m2PathRatio"] = screen_path / animation_path if animation_path > 0.5 else math.nan
                            screen_peak = max_step(candidate.screen, release, release + 10)
                            animation_peak = max_step(candidate.animation, release, release + 10)
                            row["m2PeakRatio"] = screen_peak / animation_peak if animation_peak > 1e-6 else math.nan
                        group = "loop" if meta["context"] == "loop" else "stop"
                        aggregate["m1_" + group].append(row["m1GapCm"])
                        aggregate["m2_" + group].append(row["m2PathRatio"])
                        aggregate["m2peak_" + group].append(row["m2PeakRatio"])
                event_rows.append(row)

            if case.gait == "Walk" and side == "r":
                bake = case.stop_bake(case.stop_end)
                if bake == "Bake_WalkFwdStop_RU" and candidate.screen[case.stop_end] is not None:
                    gap = distance2(candidate.screen[case.stop_end], candidate.animation[case.stop_end]) * 100.0
                    late = min(case.stop_end, case.stop_start + 40)
                    screen_path = path_length(candidate.screen, late, case.stop_end)
                    animation_path = path_length(candidate.animation, late, case.stop_end)
                    ratio = screen_path / animation_path if animation_path > 0.5 else math.nan
                    aggregate["m3gap"].append(gap)
                    aggregate["m3ratio"].append(ratio)
                    relevant = [row for row in event_rows if row["policy"] == policy.name and row["case"] == case.name
                                and row["side"] == "R" and row["context"] != "loop"]
                    if relevant:
                        target = relevant[-1]
                        target["m3EndGapCm"] = gap
                        target["m3LatePathRatio"] = ratio
                        target["m3ScreenPathCm"] = screen_path
                        target["m3AnimPathCm"] = animation_path
    return event_rows, aggregate


def correctness(cases, r0_results):
    reports = []
    all_phase_total = all_phase_match = 0
    all_residuals = []
    mismatches = []
    residual_outliers = []
    for case in cases:
        phase_total = phase_match = 0
        residuals = []
        for side in ("l", "r"):
            replay = r0_results[case.name][side]
            for i in range(20, len(case.rows)):
                expected = int(float(case.rows[i][side + "ph"]))
                actual = replay.phase[i]
                phase_total += 1
                if expected == actual:
                    phase_match += 1
                else:
                    mismatches.append((case.name, side.upper(), i, case.rows[i]["frame"], expected, actual))
                target = (number(case.rows[i], side + "tgtX"), number(case.rows[i], side + "tgtZ"))
                residual = distance2(replay.horizontal[i], target) * 100.0
                residuals.append(residual)
                if residual > 0.1:
                    residual_outliers.append((case.name, side.upper(), i, case.rows[i]["frame"], residual))
        reports.append({
            "case": case.name,
            "phase_total": phase_total,
            "phase_match": phase_match,
            "phase_rate": phase_match / phase_total if phase_total else math.nan,
            "median": median(residuals),
            "p99": percentile(residuals, 0.99),
            "max": maximum(residuals),
        })
        all_phase_total += phase_total
        all_phase_match += phase_match
        all_residuals.extend(residuals)
    overall = {
        "case": "OVERALL",
        "phase_total": all_phase_total,
        "phase_match": all_phase_match,
        "phase_rate": all_phase_match / all_phase_total if all_phase_total else math.nan,
        "median": median(all_residuals),
        "p99": percentile(all_residuals, 0.99),
        "max": maximum(all_residuals),
    }
    passed = overall["phase_rate"] >= 0.995 and overall["p99"] <= 0.1
    return reports, overall, mismatches, residual_outliers, passed


def write_events(path, rows):
    with path.open("w", encoding="utf-8", newline="") as stream:
        writer = csv.DictWriter(stream, fieldnames=EVENT_FIELDS)
        writer.writeheader()
        for row in rows:
            writer.writerow({key: fmt(value, 6) if isinstance(value, float) else value for key, value in row.items()})


def summary_row(policy, values):
    return [
        policy,
        md(median(values["m1_loop"])), md(median(values["m1_stop"])),
        md(median(values["m2_loop"])), md(maximum(values["m2_loop"])),
        md(median(values["m2_stop"])), md(maximum(values["m2_stop"])),
        md(median(values["m2peak_loop"])), md(maximum(values["m2peak_loop"])),
        md(median(values["m2peak_stop"])), md(maximum(values["m2peak_stop"])),
        md(median(values["m4_loop"])), md(maximum(values["m4_loop"])),
        md(median(values["m4_stop"])), md(maximum(values["m4_stop"])),
        md(median(values["m4peak_loop"])), md(maximum(values["m4peak_loop"])),
        md(median(values["m4peak_stop"])), md(maximum(values["m4peak_stop"])),
        md(median(values["m5"])), md(maximum(values["m5"])),
        str(values["lock_count"]), str(values["locked_frames"]),
        md(median(values["lock_lengths"])), md(maximum(values["lock_lengths"])),
        md(percentile(values["jerk"], 0.95)), md(maximum(values["jerk"])),
        md(median(values["m3gap"])), md(median(values["m3ratio"])),
    ]

def write_summary(path, reports, overall, mismatches, residual_outliers, passed, policy_summaries):
    lines = ["# T3 offline replay summary", "", "## X Bot FootIKSettings", ""]
    for key, value in SETTINGS.items():
        lines.append("- `%s = %s`" % (key, ("%g" % value)))
    lines.extend(["", "## R0 correctness (first 20 rows of each case excluded)", "",
                  "| case | phase match | phase rate | horizontal residual median cm | p99 cm | max cm |",
                  "|---|---:|---:|---:|---:|---:|"])
    for report in reports + [overall]:
        lines.append("| %s | %d/%d | %.3f%% | %s | %s | %s |" % (
            report["case"], report["phase_match"], report["phase_total"], report["phase_rate"] * 100.0,
            md(report["median"], 4), md(report["p99"], 4), md(report["max"], 4)))
    lines.extend(["", "Result: **%s** (required: phase ≥ 99.5%% and horizontal p99 ≤ 0.1 cm)." % ("PASS" if passed else "FAIL"), "",
                  "### Phase mismatch frames", ""])
    if mismatches:
        lines.append("| case | side | row | frame | recorded | replay |")
        lines.append("|---|---:|---:|---:|---:|---:|")
        for item in mismatches:
            lines.append("| %s | %s | %d | %s | %d | %d |" % item)
    else:
        lines.append("None.")
    lines.extend(["", "### Horizontal residual frames over 0.1 cm", ""])
    if residual_outliers:
        lines.append("| case | side | row | frame | residual cm |")
        lines.append("|---|---:|---:|---:|---:|")
        for case, side, row, frame, residual in residual_outliers:
            lines.append("| %s | %s | %d | %s | %.4f |" % (case, side, row, frame, residual))
    else:
        lines.append("None.")
    if passed:
        lines.extend(["", "## Policy summary", "", "| " + " | ".join(SUMMARY_HEADERS) + " |",
                      "|" + "|".join("---" if i == 0 else "---:" for i in range(len(SUMMARY_HEADERS))) + "|"])
        for policy, values in policy_summaries:
            lines.append("| " + " | ".join(summary_row(policy, values)) + " |")
        lines.extend(["", "M2 columns are screen/animation ratios. M4 is measured only on frames that R0 classified Locked. "
                      "M5 uses the `lockev.py.txt` projection definition. M7 is horizontal second difference per frame. "
                      "M3 is Walk `Bake_WalkFwdStop_RU`, right foot, at the last pre-restart row and over tRel ≥ 40."])
    else:
        lines.extend(["", "Candidate policies were not evaluated because R0 correctness failed."])
    path.write_text("\n".join(lines) + "\n", encoding="utf-8")


def parse_args(argv):
    parser = argparse.ArgumentParser(description="Replay T3 from newly recorded snap_on_*.csv files")
    parser.add_argument("input_dir", nargs="?", type=Path, help="directory containing snap_on_*.csv")
    parser.add_argument("--output-dir", type=Path, help="output directory (default: input_dir)")
    parser.add_argument("--policy", help="run R0 plus one expanded policy id")
    parser.add_argument("--combo", action="append", default=[], metavar="R2x:R1y", help="add an explicitly selected R2+R1 combo")
    parser.add_argument("--list-policies", action="store_true", help="print expanded policy ids and exit")
    return parser.parse_args(argv)


def main(argv=None):
    args = parse_args(argv)
    registry = policy_registry()
    if args.list_policies:
        for name in registry:
            print(name)
        return 0
    if args.input_dir is None:
        print("ERROR: input_dir is required unless --list-policies is used", file=sys.stderr)
        return 2
    try:
        cases = load_cases(args.input_dir)
        if args.policy and args.policy not in registry:
            raise ValueError("unknown policy %r; use --list-policies" % args.policy)
        combos = [combine_policy(spec, registry) for spec in args.combo]
    except (OSError, ValueError) as exc:
        print("ERROR: %s" % exc, file=sys.stderr)
        return 2

    output_dir = args.output_dir or args.input_dir
    output_dir.mkdir(parents=True, exist_ok=True)
    r0 = {case.name: {side: replay_foot(case, side, registry["R0"]) for side in ("l", "r")} for case in cases}
    reports, overall, mismatches, residual_outliers, passed = correctness(cases, r0)
    policy_summaries = []
    all_events = []
    if passed:
        if args.policy:
            selected = [registry["R0"]] if args.policy == "R0" else [registry["R0"], registry[args.policy]]
        else:
            selected = list(registry.values())
        selected.extend(combos)
        seen = set()
        selected = [policy for policy in selected if not (policy.name in seen or seen.add(policy.name))]
        for policy in selected:
            rows, values = evaluate_policy(cases, r0, policy)
            all_events.extend(rows)
            policy_summaries.append((policy.name, values))
    write_events(output_dir / "replay_events.csv", all_events)
    write_summary(output_dir / "replay_summary.md", reports, overall, mismatches, residual_outliers, passed, policy_summaries)
    print("R0 phase %.3f%%; horizontal residual p99 %.4f cm; %s" %
          (overall["phase_rate"] * 100.0, overall["p99"], "PASS" if passed else "FAIL"))
    print("wrote %s" % (output_dir / "replay_summary.md"))
    print("wrote %s" % (output_dir / "replay_events.csv"))
    return 0 if passed else 1


if __name__ == "__main__":
    sys.exit(main())
